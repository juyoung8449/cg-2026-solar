# 관찰하면서 발견한 불편한 점
수평 이동 중 의도하지 않은 회전이 발생함
원근감을 조절해서 카메라와 물체 사이의 거리를 변경하는 기능 자체는 비교적 잘 작동한다.
하지만 카메라를 X축 방향으로 이동하려고 해도 카메라가 함께 회전하는 현상이 발생한다.
이 때문에 물체의 수평을 유지하면서 위치만 이동하기 어렵고, 원하는 각도를 맞추기 위해 추가적인 조작이 필요하다.
수직 방향(Y축)으로 이동하기 어려움
화면 위쪽에 있는 물체를 관찰하고 싶을 때 카메라를 Y축 방향으로 직접 이동하기 어렵다.
원하는 높이로 이동하기 위해 먼저 카메라를 물체에서 멀리 떨어뜨린 뒤, 목표점을 조절하거나 Z축으로 이동한 다음 다시 X축 방향으로 조정하는 등 여러 단계를 거쳐야 한다.
따라서 단순히 위쪽에 있는 물체를 보고 싶은 상황에서도 불필요하게 여러 번 조작해야 하는 문제가 있다.
카메라의 수평을 유지하면서 이동하기 어려움
이동을 하는 과정에서 카메라가 의도하지 않게 회전하기 때문에 화면의 수평을 맞추기가 어렵다.
실제로 글씨가 있는 물체를 관찰할 때 글자를 읽을 수는 있지만, 카메라가 임의로 회전하면서 약 90도 정도 돌아간 상태에서 글자를 읽어야 하는 상황이 발생했다.
따라서 관찰 대상의 방향을 유지한 채 위치만 이동할 수 있는 기능이 필요하다.
원하는 관찰 위치까지 가는 과정이 번거로움
현재 조작 방식에서는 특정 물체를 바로 찾아가는 것보다 카메라와 목표점 사이의 거리와 방향을 여러 번 조절하면서 위치를 맞춰야 한다.
특히 위쪽이나 특정 방향에 있는 물체를 관찰할 때 이러한 문제가 크게 느껴진다.
휠 조작과 기본 조작 횟수의 관계가 불편함
마우스 휠을 한 번 굴리는 동작도 기본 조작 횟수로 카운트되는 것처럼 보여, 휠을 많이 사용해서 원하는 위치를 찾으면 조작 횟수가 불필요하게 증가한다.
따라서 연속적인 휠 조작을 각각의 독립적인 조작으로 계산하기보다는, 실제 의미 있는 카메라 조작 단위로 기록하는 방식이 더 적절할 것 같다.

# 1. 사용자 및 관찰 목표
사용자

본 시스템의 사용자는 3D 공간에 배치된 물체를 관찰하기 위해 카메라를 이동시키는 사용자이다. 특히 특정 물체의 위치를 확인하거나 물체에 표시된 글자와 방향을 자세히 관찰해야 하므로, 카메라의 위치와 방향을 직관적으로 조작할 수 있어야 한다.

관찰 목표

기존 조작 방식에서는 카메라를 이동하는 과정에서 의도하지 않은 회전이 발생하고, Y축 방향으로 이동하기 어려워 원하는 관찰 위치에 도달하기 위해 여러 번의 조작이 필요했다. 또한 카메라가 기울어져 관찰 대상의 수평을 유지하기 어려운 문제가 있었다.

따라서 개선된 조작의 목표를 다음과 같이 설정하였다.

카메라를 이동할 때 의도하지 않은 회전이 발생하지 않도록 한다.
X축과 Y축 방향으로 카메라를 직접 이동할 수 있도록 한다.
카메라의 수평을 유지하여 물체를 안정적인 방향에서 관찰할 수 있도록 한다.
특정 방향에 있는 물체를 적은 조작으로 찾아갈 수 있도록 한다.
필요한 경우에는 별도의 조작으로 카메라 방향만 회전할 수 있도록 한다.
연속적인 마우스 휠 조작이 불필요하게 많은 조작 횟수로 기록되지 않도록 한다.

즉, 평소에는 이동만 수행하고, 사용자가 명시적으로 Alt를 누른 경우에만 카메라 방향을 변경한다.


#대표코드1 (카메라 위치 계산)
function camera() {
  // 카메라 회전(rotation)을 적용한 Z축 거리(distance) 오프셋 계산
  const offset = rotate(
    state.rotation,
    [0, 0, state.distance]
  );

  return {
    // 카메라 실제 위치 = 시점 타깃 + 오프셋
    eye: state.target.map((v, i) => v + offset[i]),
    
    // 카메라가 바라보는 중심점
    target: [...state.target],
    
    // 카메라의 상방 벡터 (수평 유지)
    up: rotate(state.rotation, [0, 1, 0])
  };
}

#대표코드2 (마우스 드래그 및 평면이동)
canvas.addEventListener('pointermove', e => {
  if (!drag || drag.id !== e.pointerId) return;

  const r = canvas.getBoundingClientRect();

  // 화면 높이 및 FOV, 거리에 비례한 월드 좌표 이동 단위 계산
  const unit =
    2 *
    state.distance *
    Math.tan(state.fov * Math.PI / 360) /
    r.height;

  const dx = e.clientX - drag.x;
  const dy = e.clientY - drag.y;

  // X축 이동: 오른쪽 드래그 (+dx) -> 오른쪽 (+X) 이동
  state.target[0] = drag.target[0] + dx * unit;

  // Y축 이동 (반대 설정): 위로 드래그 (-dy) -> 아래 (-Y) 이동
  //                         아래로 드래그 (+dy) -> 위 (+Y) 이동
  state.target[1] = drag.target[1] + dy * unit;
});

// 마우스 휠 줌 (거리 조절)
canvas.addEventListener('wheel', e => {
  e.preventDefault();

  // 거리 한계점 제한 (0.12 ~ 100)
  state.distance = Math.max(
    0.12,
    Math.min(
      100,
      state.distance * Math.exp(e.deltaY * 0.001)
    )
  );

  if (wheelTimer === null) state.actions++;
  clearTimeout(wheelTimer);
  wheelTimer = setTimeout(() => { wheelTimer = null; }, 250);
}, { passive: false });

// Home 키 입력 시 초기화
canvas.addEventListener('keydown', e => {
  if (e.key !== 'Home') return;
  e.preventDefault();
  state.actions++;
  home(); // 초기 위치로 복구
});
